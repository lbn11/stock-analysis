### 步骤2: 修改 index.html

在 `<script>` 标签开始处添加配置：

```javascript
// API配置
const API_CONFIG = {
    USE_REAL_DATA: true,  // true=使用真实数据, false=使用模拟数据
    API_BASE_URL: 'http://localhost:5000/api'  // 后端API地址
};
```

### 步骤3: 修改 analyzeStock 函数

```javascript
async function analyzeStock() {
    const stockCode = document.getElementById('stockCode').value;
    const analysisType = document.getElementById('analysisType').value;
    const timeHorizon = document.getElementById('timeHorizon').value;
    const loading = document.getElementById('loading');
    const results = document.getElementById('results');
    const btn = document.querySelector('.analyze-btn');

    if (!stockCode) {
        alert('请输入股票代码');
        return;
    }

    loading.classList.add('active');
    results.classList.remove('active');
    btn.disabled = true;
    btn.textContent = '⏳ 分析中...';

    try {
        let analysisData;

        if (API_CONFIG.USE_REAL_DATA) {
            // 调用真实API
            const response = await fetch(`${API_CONFIG.API_BASE_URL}/analyze`, {
                method: 'POST',
                headers: {
                    'Content-Type': 'application/json'
                },
                body: JSON.stringify({
                    stock_code: stockCode,
                    analysis_type: analysisType,
                    time_horizon: timeHorizon
                })
            });

            if (!response.ok) {
                throw new Error('API请求失败');
            }

            analysisData = await response.json();
        } else {
            // 使用模拟数据
            await new Promise(resolve => setTimeout(resolve, 2000));
            analysisData = generateMockData(stockCode, analysisType, timeHorizon);
        }

        displayResults(analysisData, analysisType);

        loading.classList.remove('active');
        results.classList.add('active');
        btn.disabled = false;
        btn.textContent = '🔍 开始分析';

    } catch (error) {
        console.error('错误:', error);
        alert('获取数据失败：' + error.message + '\n将使用模拟数据');

        // 失败时回退到模拟数据
        const analysisData = generateMockData(stockCode, analysisType, timeHorizon);
        displayResults(analysisData, analysisType);

        loading.classList.remove('active');
        results.classList.add('active');
        btn.disabled = false;
        btn.textContent = '🔍 开始分析';
    }
}
```

### ⚠️ 跨域问题解决

由于浏览器CORS限制，直接调用第三方API会失败。解决方案：

#### 解决方案1: 使用CORS代理
```javascript
// 使用公共CORS代理
async function fetchWithProxy(url) {
    const proxyUrl = `https://api.allorigins.win/raw?url=${encodeURIComponent(url)}`;
    const response = await fetch(proxyUrl);
    return await response.json();
}
```

#### 解决方案2: 使用JSONP（如果API支持）
```javascript
function fetchWithJSONP(url, callback) {
    const script = document.createElement('script');
    script.src = `${url}&callback=${callback}`;
    document.body.appendChild(script);
}
```
